rules_version = '2';

service cloud.firestore {

  match /databases/{database}/documents {

    // ============================================================
    // ADMIN
    // ============================================================

    function isAdmin() {
      return request.auth != null
             && request.auth.token.email == "jonvitor014@gmail.com";
    }


    // ============================================================
    // USERS
    // ============================================================

    match /users/{userId} {

      // Usuário pode ler o próprio perfil.
      // Admin pode ler todos os usuários.
      allow read: if request.auth != null
                  && (
                    isAdmin()
                    || request.auth.uid == userId
                  );

      // Usuário pode alterar o próprio perfil.
      // Admin pode alterar qualquer usuário.
      allow write: if request.auth != null
                   && (
                     isAdmin()
                     || request.auth.uid == userId
                   );

      // ============================================================
      // USER PROJECTS (Cross-Device Sync)
      // ============================================================
      
      match /projects/{projectId} {
        // Usuário pode ler seus próprios projetos
        // Admin pode ler todos os projetos
        allow read: if request.auth != null
                    && (
                      isAdmin()
                      || request.auth.uid == userId
                    );

        // Usuário pode criar/alterar/excluir seus próprios projetos
        // Admin pode gerenciar qualquer projeto
        allow write: if request.auth != null
                     && (
                       isAdmin()
                       || request.auth.uid == userId
                     );
      }

      // ============================================================
      // USER PLAYLISTS (Cross-Device Sync)
      // ============================================================
      
      match /playlists/{playlistId} {
        // Usuário pode ler suas próprias playlists
        // Admin pode ler todas as playlists
        allow read: if request.auth != null
                    && (
                      isAdmin()
                      || request.auth.uid == userId
                    );

        // Usuário pode criar/alterar/excluir suas próprias playlists
        // Admin pode gerenciar qualquer playlist
        allow write: if request.auth != null
                     && (
                       isAdmin()
                       || request.auth.uid == userId
                     );
      }
    }


    // ============================================================
    // USUARIOS
    // ============================================================

    match /usuarios/{userId} {

      allow read: if request.auth != null;

      allow write: if request.auth != null;

      match /projetos/{projectId} {

        allow read: if request.auth != null
                    && request.auth.uid == userId;

        allow write: if request.auth != null
                     && request.auth.uid == userId;
      }
    }


    // ============================================================
    // CREATORS
    // ============================================================

    match /creators/{creatorId} {

      // Público para leitura.
      allow read: if true;

      // Dono pode alterar o próprio creator.
      // Admin pode alterar qualquer creator.
      allow write: if request.auth != null
                   && (
                     isAdmin()
                     || request.auth.uid == creatorId
                   );
    }


    // ============================================================
    // COMMUNITY TRACKS
    // ============================================================

    match /communityTracks/{trackId} {

      // Qualquer pessoa pode visualizar tracks.
      allow read: if true;

      // Usuário autenticado pode criar.
      allow create: if request.auth != null;

      // Usuário autenticado pode editar.
      allow update: if request.auth != null;

      // Usuário autenticado pode excluir.
      allow delete: if request.auth != null;
    }


    // ============================================================
    // WAITLIST
    // ============================================================

    match /waitlist/{waitlistId} {

      allow read: if request.auth != null;

      allow write: if request.auth != null;
    }


    // ============================================================
    // SUPPORT REQUESTS
    // ============================================================

    match /supportRequests/{supportId} {

      // Usuário pode visualizar a própria solicitação.
      // Admin pode visualizar todas as solicitações.
      allow read: if request.auth != null
                  && (
                    isAdmin()
                    || resource.data.userId == request.auth.uid
                  );

      // Usuário autenticado pode criar solicitação.
      allow create: if request.auth != null;

      // Usuário pode alterar/excluir a própria solicitação.
      // Admin pode alterar/excluir qualquer solicitação.
      allow update, delete: if request.auth != null
                            && (
                              isAdmin()
                              || resource.data.userId == request.auth.uid
                            );
    }


    // ============================================================
    // COMPLAINTS
    // ============================================================

    match /complaints/{complaintId} {

      // Usuário vê a própria reclamação.
      // Admin vê todas.
      allow read: if request.auth != null
                  && (
                    isAdmin()
                    || resource.data.userId == request.auth.uid
                  );

      // Usuário autenticado pode criar reclamação.
      allow create: if request.auth != null;

      // Usuário pode alterar/excluir a própria reclamação.
      // Admin pode alterar/excluir qualquer reclamação.
      allow update, delete: if request.auth != null
                            && (
                              isAdmin()
                              || resource.data.userId == request.auth.uid
                            );
    }


    // ============================================================
    // SUPPORT MESSAGES
    // ============================================================

    match /supportMessages/{messageId} {

      // Usuário vê as próprias mensagens.
      // Admin pode visualizar todas as mensagens.
      allow read: if request.auth != null
                  && (
                    isAdmin()
                    || resource.data.userId == request.auth.uid
                  );

      // Usuário autenticado pode enviar mensagem.
      allow create: if request.auth != null;

      // Usuário pode alterar/excluir a própria mensagem.
      // Admin pode alterar/excluir qualquer mensagem.
      allow update, delete: if request.auth != null
                            && (
                              isAdmin()
                              || resource.data.userId == request.auth.uid
                            );
    }


    // ============================================================
    // USER FAVORITES
    // ============================================================

    match /userFavorites/{userId} {

      // Usuário vê os próprios favoritos.
      allow read: if request.auth != null
                  && request.auth.uid == userId;

      // Usuário altera os próprios favoritos.
      allow write: if request.auth != null
                   && request.auth.uid == userId;
    }


    // ============================================================
    // GLOBAL CONFIG
    // ============================================================

    match /global_config/{configId} {

      // Usuários autenticados podem ler.
      allow read: if request.auth != null;

      // Apenas admin pode escrever (para configurações de checkout, etc).
      allow write: if isAdmin();
    }


    // ============================================================
    // SETLISTS
    // ============================================================

    match /setlists/{setlistId} {

      // Dono pode ler.
      allow read: if request.auth != null
                  && resource.data.userId == request.auth.uid;

      // Usuário autenticado pode criar.
      allow create: if request.auth != null;

      // Dono pode alterar/excluir.
      allow update, delete: if request.auth != null
                            && resource.data.userId == request.auth.uid;
    }


    // ============================================================
    // PROJECTS (Legacy - Mantido para compatibilidade)
    // ============================================================

    match /projects/{projectId} {

      // Dono pode ler.
      allow read: if request.auth != null
                  && resource.data.userId == request.auth.uid;

      // Usuário autenticado pode criar.
      allow create: if request.auth != null;

      // Dono pode alterar/excluir.
      allow update, delete: if request.auth != null
                            && resource.data.userId == request.auth.uid;
    }


    // ============================================================
    // PUBLIC SETLISTS
    // ============================================================

    match /publicSetlists/{shareId} {

      // Público para leitura.
      allow read: if true;

      // Apenas autenticados podem criar.
      allow create: if request.auth != null;

      // Dono pode editar/excluir.
      allow update, delete: if request.auth != null
                            && resource.data.userId == request.auth.uid;
    }


    // ============================================================
    // PAYMENT TOKENS
    // ============================================================

    match /paymentTokens/{tokenId} {

      // Usuário pode ler o próprio token.
      // Admin pode ler todos os tokens.
      allow read: if request.auth != null
                  && (
                    isAdmin()
                    || resource.data.uid == request.auth.uid
                  );

      // Usuário autenticado pode criar tokens se o uid bater.
      // Admin pode criar tokens para qualquer usuário.
      allow create: if request.auth != null
                    && (
                      isAdmin()
                      || request.resource.data.uid == request.auth.uid
                    );

      // Usuário pode atualizar o próprio token (para transações).
      // Admin pode atualizar qualquer token.
      allow update: if request.auth != null
                    && (
                      isAdmin()
                      || resource.data.uid == request.auth.uid
                    );

      // Usuário pode deletar o próprio token.
      // Admin pode deletar qualquer token.
      allow delete: if request.auth != null
                     && (
                       isAdmin()
                       || resource.data.uid == request.auth.uid
                     );
    }


    // ============================================================
    // TRIAL FORMS
    // ============================================================

    match /trialForms/{formId} {

      // Admin pode ler todos os formulários.
      // Usuário pode ler o próprio formulário (se o uid bater).
      allow read: if request.auth != null
                  && (
                    isAdmin()
                    || resource.data.uid == request.auth.uid
                  );

      // Usuário autenticado pode criar formulário com seu próprio UID.
      // Admin pode criar formulários para qualquer usuário.
      allow create: if request.auth != null
                    && (
                      isAdmin()
                      || request.resource.data.uid == request.auth.uid
                    );

      // Usuário pode atualizar o próprio formulário.
      // Admin pode atualizar qualquer formulário.
      // Verifica tanto o documento existente quanto o novo para impedir troca de proprietário.
      allow update: if request.auth != null
                    && (
                      isAdmin()
                      || (
                        resource.data.uid == request.auth.uid
                        && request.resource.data.uid == request.auth.uid
                      )
                    );

      // Admin pode excluir qualquer formulário.
      // Usuário pode excluir o próprio formulário.
      allow delete: if request.auth != null
                     && (
                       isAdmin()
                       || resource.data.uid == request.auth.uid
                     );
    }


    // ============================================================
    // NEWS (Notícias e Enquetes)
    // ============================================================

    match /news/{newsId} {

      // Usuários autenticados podem ler notícias e enquetes.
      allow read: if request.auth != null;

      // Apenas admin pode criar notícias e enquetes.
      allow create: if isAdmin();

      // Usuários autenticados podem atualizar (curtir/votar).
      // Admin pode atualizar qualquer notícia.
      allow update: if request.auth != null
                   && (
                     isAdmin()
                     || (
                       // Permitir curtidas (adicionar/remover uid do array likes)
                       request.resource.data.likes == resource.data.likes + [request.auth.uid]
                       || request.resource.data.likes == resource.data.likes.diff([request.auth.uid])
                       // Permitir votos em enquetes
                       || request.resource.data.votedBy == resource.data.votedBy + [request.auth.uid]
                       // Permitir atualização de opções de votação (admin)
                       || isAdmin()
                     )
                   );

      // Apenas admin pode excluir notícias e enquetes.
      allow delete: if isAdmin();
    }


    // ============================================================
    // ADMINS (Verificação de Admin para Painel da Equipe)
    // ============================================================

    match /admins/{adminId} {

      // Admin pode ler sua própria entrada na coleção admins.
      // Isso permite que o painel da equipe verifique se um usuário é admin.
      allow read: if request.auth != null
                  && (
                    isAdmin()
                    || request.auth.uid == adminId
                  );

      // Apenas admin (definido pela função isAdmin) pode criar/alterar admins.
      allow write: if isAdmin();
    }


    // ============================================================
    // SUBSCRIPTION HISTORY (Histórico de Assinaturas)
    // ============================================================

    match /subscriptionHistory/{historyId} {
      // Admin pode ler todos os históricos
      allow read: if isAdmin();
      
      // Admin pode criar novos registros
      allow create: if isAdmin();
      
      // Bloqueia escrita (os registros são imutáveis)
      allow update, delete: if false;
    }

    // ============================================================
    // TRIAL TOKENS
    // ============================================================

    match /trialTokens/{tokenId} {
      allow read: if isAdmin();
      allow create: if isAdmin();
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }

    // ============================================================
    // BLOQUEIO PADRÃO
    // ============================================================

    // Qualquer coleção não especificada acima fica bloqueada.
    match /{document=**} {

      allow read, write: if false;
    }
  }
}
